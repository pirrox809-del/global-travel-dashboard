   211→function AttachmentGalleryModal({
   212→  attachment,
   213→  originRect,
   214→  onClose,
   215→}: {
   216→  attachment: Attachment;
   217→  originRect: DOMRect;
   218→  onClose: () => void;
   219→}) {
   220→  const [phase, setPhase] = useState<"opening" | "open" | "closing">("opening");
   221→  const [targetRect, setTargetRect] = useState<{
   222→    top: number;
   223→    left: number;
   224→    width: number;
   225→    height: number;
   226→    radius: number;
   227→  } | null>(null);
   228→  const imgRef = useRef<HTMLImageElement>(null);
   229→
   230→  useEffect(() => {
   231→    const maxW = Math.min(window.innerWidth * 0.86, 560);
   232→    const maxH = Math.min(window.innerHeight * 0.78, 720);
   233→
   234→    const naturalW = attachment.width || 800;
   235→    const naturalH = attachment.height || 600;
   236→    const scale = Math.min(maxW / naturalW, maxH / naturalH, 1.6);
   237→
   238→    const width = naturalW * scale;
   239→    const height = naturalH * scale;
   240→
   241→    setTargetRect({
   242→      top: (window.innerHeight - height) / 2,
   243→      left: (window.innerWidth - width) / 2,
   244→      width,
   245→      height,
   246→      radius: 20,
   247→    });
   248→
   249→    const raf = requestAnimationFrame(() => setPhase("open"));
   250→    return () => cancelAnimationFrame(raf);
   251→  }, [attachment]);
   252→
   253→  const handleClose = useCallback(() => setPhase("closing"), []);
   254→
   255→  useEffect(() => {
   256→    const onKey = (e: KeyboardEvent) => { if (e.key === "Escape") handleClose(); };
   257→    document.addEventListener("keydown", onKey);
   258→    return () => document.removeEventListener("keydown", onKey);
   259→  }, [handleClose]);
   260→
   261→  const isOpen = phase === "open";
   262→  const isClosing = phase === "closing";
   263→
   264→  const geometry = isOpen && targetRect
   265→      ? targetRect
   266→      : { top: originRect.top, left: originRect.left, width: originRect.width, height: originRect.height, radius: 12 };
   267→
   268→  const animEasing = isClosing ? "ease-out" : "cubic-bezier(0.175, 0.885, 0.32, 1.275)";
   269→  const animDur = isClosing ? "0.3s" : "0.45s";
   270→  const flipTransition = `top ${animDur} ${animEasing}, left ${animDur} ${animEasing}, width ${animDur} ${animEasing}, height ${animDur} ${animEasing}, border-radius ${animDur} ${animEasing}`;
   271→
   272→  return (
   273→    <div className="fixed inset-0 z-[100]" onClick={handleClose} role="dialog" aria-modal="true">
   274→      <div className="absolute inset-0 bg-background/70 backdrop-blur-md transition-opacity duration-400" style={{ opacity: isOpen ? 1 : 0 }} />
   275→      <div
   276→        style={{
   277→          position: "fixed",
   278→          top: geometry.top, left: geometry.left, width: geometry.width, height: geometry.height,
   279→          borderRadius: geometry.radius, transition: flipTransition, overflow: "hidden",
   280→          boxShadow: isOpen ? "0 24px 60px -12px rgb(0 0 0 / 0.35)" : "0 0px 0px 0px rgb(0 0 0 / 0)",
   281→        }}
   282→        className="bg-muted"
   283→        onTransitionEnd={() => { if (phase === "closing") onClose(); }}
   284→        onClick={(e) => e.stopPropagation()}
   285→      >
   286→        <img ref={imgRef} src={attachment.url} alt={attachment.name} className="size-full object-cover" draggable={false} />
   287→      </div>

   289→      <button
   290→        type="button" onClick={handleClose}
   291→        style={{ opacity: isOpen ? 1 : 0, transform: isOpen ? "scale(1)" : "scale(0.7)" }}
   292→        className={cn(
   293→          "fixed right-4 top-4 flex size-9 items-center justify-center rounded-full bg-card/90 text-foreground/70 shadow-md backdrop-blur-sm",
   294→          "transition-all duration-300 ease-[cubic-bezier(0.175,0.885,0.32,1.275)] hover:bg-card hover:text-foreground",
   295→          !isOpen && "pointer-events-none"
   296→        )}
   297→      >
   298→        <span className="scale-150"><CloseIcon /></span>
   299→      </button>
   300→    </div>
   301→  );
   302→}
