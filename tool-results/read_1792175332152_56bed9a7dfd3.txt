     1→/**
     2→ * i18n.ts — Lightweight internationalization dictionaries.
     3→ *
     4→ * ---------------------------------------------------------------------
     5→ * Why a custom solution (not next-intl routing)?
     6→ * ---------------------------------------------------------------------
     7→ * The project constraint is "user can only see the / route". next-intl's
     8→ * standard setup uses a `[locale]` URL segment (/en, /it, …) which would
     9→ * create multiple routes. Instead we use a Context-based locale (no URL
    10→ * change) + a `?lang=` query param sent to the API so the server knows
    11→ * which language to generate LLM content in.
    12→ *
    13→ * 4 locales: English (default), Spanish, Italian, French.
    14→ * Add a language: add a key to `LOCALES` + a dictionary to `MESSAGES`.
    15→ */

   531→export type MessageKey = keyof Dict;

   533→/** Region name translation (closed set: ~6 region values from the dataset). */
   534→const REGION_TRANSLATIONS: Record<Locale, Record<string, string>> = {
   535→  en: { Africa: 'Africa', Americas: 'Americas', Asia: 'Asia', Europe: 'Europe', Oceania: 'Oceania', Antarctic: 'Antarctic' },
   536→  es: { Africa: 'África', Americas: 'Américas', Asia: 'Asia', Europe: 'Europa', Oceania: 'Oceanía', Antarctic: 'Antártida' },
   537→  it: { Africa: 'Africa', Americas: 'Americhe', Asia: 'Asia', Europe: 'Europa', Oceania: 'Oceania', Antarctic: 'Antartide' },
   538→  fr: { Africa: 'Afrique', Americas: 'Amériques', Asia: 'Asie', Europe: 'Europe', Oceania: 'Océanie', Antarctic: 'Antarctique' },
   539→};
   540→
   541→/** Translate a region name. Falls back to the original if unknown. */
   542→export function translateRegion(region: string, locale: Locale): string {
   543→  return REGION_TRANSLATIONS[locale]?.[region] ?? region;
   544→}
   545→
   546→/** Format a weekday+day date in the given locale. */
   547→export function formatDate(weekdayDay: string, locale: Locale): string {
   548→  try {
   549→    return new Date(weekdayDay).toLocaleDateString(locale === 'en' ? 'en-US' : locale, {
   550→      weekday: 'short',
   551→      day: 'numeric',
   552→    });
   553→  } catch {
   554→    return weekdayDay;
   555→  }
   556→}
   557→
   558→/** Format a full date (e.g. advisory "updated" date) in the given locale. */
   559→export function formatDateLong(dateStr: string, locale: Locale): string {
   560→  try {
   561→    return new Date(dateStr).toLocaleDateString(locale === 'en' ? 'en-US' : locale, {
   562→      year: 'numeric',
   563→      month: 'short',
   564→      day: 'numeric',
   565→    });
   566→  } catch {
   567→    return dateStr;
   568→  }
   569→}
   570→
   571→/** Format a time using the locale's convention. */
   572→export function formatTime(timeStr: string, locale: Locale): string {
   573→  try {
   574→    return new Date(timeStr).toLocaleTimeString(locale === 'en' ? 'en-US' : locale);
   575→  } catch {
   576→    return timeStr;
   577→  }
   578→}

   580→/**
   581→ * Resolve a translation key with optional {placeholder} interpolation.
   582→ * Falls back to English, then to the key itself (so missing strings are
   583→ * visible during dev rather than silently empty).
   584→ */
   585→export function translate(locale: Locale, key: MessageKey, params?: Record<string, string | number>): string {
   586→  let str = MESSAGES[locale]?.[key] ?? MESSAGES.en[key] ?? key;
   587→  if (params) {
   588→    for (const [k, v] of Object.entries(params)) {
   589→      str = str.replace(`{${k}}`, String(v));
   590→    }
   591→  }
   592→  return str;
   593→}
